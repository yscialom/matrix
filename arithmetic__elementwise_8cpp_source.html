112// Helper concepts — GCC does not support bare requires-expressions in static_assert
113// when all candidates are constrained out (same workaround as in concepts.cpp).
114// Note: operator+ and operator- delegate to +=/−=, so testing += and -= is sufficient.
115template <class M>
116concept matrix_add_assignable = requires(M& a, const M& b) { a += b; };
117template <class M>
118concept matrix_sub_assignable = requires(M& a, const M& b) { a -= b; };
119
120static_assert(!matrix_add_assignable<ysc::matrix<no_arithmetic, 2>>);
121static_assert(!matrix_sub_assignable<ysc::matrix<no_arithmetic, 2>>);
122
123static_assert(matrix_add_assignable<ysc::matrix<int, 2>>);
124static_assert(matrix_sub_assignable<ysc::matrix<int, 2>>);
