74// Print a hyperslice starting at `it`, covering dimension `dim_idx` onward.
75// Returns an iterator past the last element printed.
76// NOLINTNEXTLINE(misc-no-recursion)
77template <class It, class Dims>
78It print_recursive(std::ostream& os, It it, const Dims& dims, std::size_t dim_idx) {
79 os << '[';
80 const std::size_t count = dims[dim_idx];
81 const bool last_dim = (dim_idx + 1 == dims.size());
82 for (std::size_t i = 0; i < count; ++i) {
83 if (i > 0) {
84 os << ", ";
85 }
86 if (last_dim) {
87 os << *it++;
88 } else {
89 // NOLINTNEXTLINE(misc-no-recursion)
90 it = print_recursive(os, it, dims, dim_idx + 1);
91 }
92 }
93 os << ']';
94 return it;
95}

97// cache-friendly: neighbor objects within the right-most coordinate are neighbors in memory
98template <class TDim, class TCoord>
99constexpr auto coordinates_to_index(TDim const& dimensions, TCoord const& coords) {
100 // Row-major index: index = c[0]*D[1]*…*D[N-1] + c[1]*D[2]*…*D[N-1] + … + c[N-1]
101 // Evaluated right-to-left; `stride` is the product of all dimensions already visited.
102 std::size_t index = 0;
103 std::size_t stride = 1;
104 auto dim = dimensions.crbegin();
105 auto coord = coords.crbegin();
106 for (; dim != dimensions.crend(); ++dim, ++coord) {
107 index += static_cast<std::size_t>(*coord) * stride;
108 stride *= *dim;
109 }
110 return index;
111}

113// Inverse of coordinates_to_index: converts a flat row-major index to multi-dimensional
114// coordinates. Rightmost coordinate varies fastest (row-major).
115template <std::size_t N>
116constexpr std::array<std::size_t, N> index_to_coordinates(std::array<std::size_t, N> dims,
117 std::size_t k) noexcept {
118 std::array<std::size_t, N> coords{};
119 auto it_c = coords.rbegin();
120 auto it_d = dims.crbegin();
121 for (; it_c != coords.rend(); ++it_c, ++it_d) {
122 *it_c = k % *it_d;
123 k /= *it_d;
124 }
125 return coords;
126}

237 offset(std::array<std::size_t, n_specs> dims,

239 std::size_t result = 0;
