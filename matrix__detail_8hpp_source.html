73// Print a hyperslice starting at `it`, covering dimension `dim_idx` onward.
74// Returns an iterator past the last element printed.
75// NOLINTNEXTLINE(misc-no-recursion)
76template <class It, class Dims>
77It print_recursive(std::ostream& os, It it, const Dims& dims, std::size_t dim_idx) {
78 os << '[';
79 const std::size_t count = dims[dim_idx];
80 const bool last_dim = (dim_idx + 1 == dims.size());
81 for (std::size_t i = 0; i < count; ++i) {
82 if (i > 0) {
83 os << ", ";
84 }
85 if (last_dim) {
86 os << *it++;
87 } else {
88 // NOLINTNEXTLINE(misc-no-recursion)
89 it = print_recursive(os, it, dims, dim_idx + 1);
90 }
91 }
92 os << ']';
93 return it;
94}

96// cache-friendly: neighbor objects within the right-most coordinate are neighbors in memory
97template <class TDim, class TCoord>
98constexpr auto coordinates_to_index(TDim const& dimensions, TCoord const& coords) {
99 // Row-major index: index = c[0]*D[1]*…*D[N-1] + c[1]*D[2]*…*D[N-1] + … + c[N-1]
100 // Evaluated right-to-left; `stride` is the product of all dimensions already visited.
101 std::size_t index = 0;
102 std::size_t stride = 1;
103 auto dim = dimensions.crbegin();
104 auto coord = coords.crbegin();
105 for (; dim != dimensions.crend(); ++dim, ++coord) {
106 index += static_cast<std::size_t>(*coord) * stride;
107 stride *= *dim;
108 }
109 return index;
110}

112// Inverse of coordinates_to_index: converts a flat row-major index to multi-dimensional
113// coordinates. Rightmost coordinate varies fastest (row-major).
114template <std::size_t N>
115constexpr std::array<std::size_t, N> index_to_coordinates(std::array<std::size_t, N> dims,
116 std::size_t k) noexcept {
117 std::array<std::size_t, N> coords{};
118 auto it_c = coords.rbegin();
119 auto it_d = dims.crbegin();
120 for (; it_c != coords.rend(); ++it_c, ++it_d) {
121 *it_c = k % *it_d;
122 k /= *it_d;
123 }
124 return coords;
125}

210 [[nodiscard]] static constexpr std::size_t
211 // NOLINTNEXTLINE(bugprone-easily-swappable-parameters)
212 offset(std::array<std::size_t, n_specs> dims,
213 std::array<std::size_t, n_specs> spec_vals) noexcept {
214 std::size_t result = 0;
215 for (std::size_t i = 0; i < n_specs; ++i) {
216 // NOLINTNEXTLINE(cppcoreguidelines-pro-bounds-constant-array-index)
217 if (is_fixed[i]) {
218 std::size_t stride = 1;
219 for (std::size_t j = i + 1; j < n_specs; ++j) {
220 // NOLINTNEXTLINE(cppcoreguidelines-pro-bounds-constant-array-index)
221 stride *= dims[j];
222 }
223 // NOLINTNEXTLINE(cppcoreguidelines-pro-bounds-constant-array-index)
224 result += spec_vals[i] * stride;
225 }
226 }
227 return result;
228 }

239 strides(std::array<std::size_t, n_specs> dims) noexcept {
240 std::array<std::size_t, n_kept> result{};
241 std::size_t k = 0;

243 // NOLINTNEXTLINE(cppcoreguidelines-pro-bounds-constant-array-index)
