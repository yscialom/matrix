577 template <typename... Specs>
578 requires(sizeof...(Specs) <= order) &&
579 ((std::same_as<Specs, all_t> || std::integral<Specs>) && ...) &&
580 (sizeof...(Specs) < order || (std::same_as<Specs, all_t> || ...))
581 [[nodiscard]] constexpr auto slice(Specs... specs) const {
582 using PaddedT = detail::pad_right_with_all_t<order, Specs...>;
583 using KeptDims = typename detail::filter_kept_dims<PaddedT, Dimensions...>::type;
584 constexpr bool is_prefix = detail::is_prefix_slice_v<PaddedT>;
585 using Storage = std::conditional_t<is_prefix, contiguous, strided>;
586 using ViewT = detail::make_matrix_view_t<T, Storage, KeptDims>;
587
588 std::array<std::size_t, order> spec_vals{};
589 {
590 std::size_t i = 0;
591 (
592 [&](auto s) {
593 using S = std::remove_cvref_t<decltype(s)>;
594 if constexpr (!detail::is_all_v<S>) {
595 auto idx = static_cast<std::size_t>(s);
596 if (idx >= dimensions[i]) {
597 throw std::out_of_range("slice: index out of range");
598 }
599 spec_vals[i] = idx;
600 }
601 ++i;
602 }(specs),
603 ...);
604 }
605
606 auto* base = _ptr + detail::slice_helper<PaddedT>::offset(dimensions, spec_vals);
607 if constexpr (is_prefix) {
608 return ViewT{base};
609 } else {
610 return ViewT{base, detail::slice_helper<PaddedT>::strides(dimensions)};
611 }
612 }

1310constexpr matrix_view<T, contiguous, Dimensions...>::operator matrix_view<
1311 T, strided, Dimensions...>() const noexcept {
1312 constexpr std::array<std::size_t, sizeof...(Dimensions)> dims_arr = {Dimensions...};
1313 std::array<std::size_t, sizeof...(Dimensions)> strides{};
1314 for (std::size_t i = 0; i < sizeof...(Dimensions); ++i) {
1315 // NOLINTNEXTLINE(cppcoreguidelines-pro-bounds-constant-array-index)
1316 strides[i] = 1;
1317 for (std::size_t j = i + 1; j < sizeof...(Dimensions); ++j) {
1318 // NOLINTNEXTLINE(cppcoreguidelines-pro-bounds-constant-array-index)
1319 strides[i] *= dims_arr[j];
1320 }
1321 }
1322 return matrix_view<T, strided, Dimensions...>{_ptr, strides};
