570 template <typename... Specs>
571 requires(sizeof...(Specs) <= order) &&
572 ((std::same_as<Specs, all_t> || std::integral<Specs>) && ...) &&
573 (sizeof...(Specs) < order || (std::same_as<Specs, all_t> || ...))
574 [[nodiscard]] constexpr auto slice(Specs... specs) const {
575 using PaddedT = detail::pad_right_with_all_t<order, Specs...>;
576 using KeptDims = typename detail::filter_kept_dims<PaddedT, Dimensions...>::type;
577 constexpr bool is_prefix = detail::is_prefix_slice_v<PaddedT>;
578 using Storage = std::conditional_t<is_prefix, contiguous, strided>;
579 using ViewT = detail::make_matrix_view_t<T, Storage, KeptDims>;
580
581 std::array<std::size_t, order> spec_vals{};
582 {
583 std::size_t i = 0;
584 (
585 [&](auto s) {
586 using S = std::remove_cvref_t<decltype(s)>;
587 if constexpr (!detail::is_all_v<S>) {
588 auto idx = static_cast<std::size_t>(s);
589 if (idx >= dimensions[i]) {
590 throw std::out_of_range("slice: index out of range");
591 }
592 spec_vals[i] = idx;
593 }
594 ++i;
595 }(specs),
596 ...);
597 }
598
599 auto* base = _ptr + detail::slice_helper<PaddedT>::offset(dimensions, spec_vals);
600 if constexpr (is_prefix) {
601 return ViewT{base};
602 } else {
603 return ViewT{base, detail::slice_helper<PaddedT>::strides(dimensions)};
604 }
605 }
