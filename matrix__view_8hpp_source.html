573 template <typename... Specs>
574 requires(sizeof...(Specs) <= order) &&
575 ((std::same_as<Specs, all_t> || std::integral<Specs>) && ...) &&
576 (sizeof...(Specs) < order || (std::same_as<Specs, all_t> || ...))
577 [[nodiscard]] constexpr auto slice(Specs... specs) const {
578 using PaddedT = detail::pad_right_with_all_t<order, Specs...>;
579 using KeptDims = typename detail::filter_kept_dims<PaddedT, Dimensions...>::type;
580 constexpr bool is_prefix = detail::is_prefix_slice_v<PaddedT>;
581 using Storage = std::conditional_t<is_prefix, contiguous, strided>;
582 using ViewT = detail::make_matrix_view_t<T, Storage, KeptDims>;
583
584 std::array<std::size_t, order> spec_vals{};
585 {
586 std::size_t i = 0;
587 (
588 [&](auto s) {
589 using S = std::remove_cvref_t<decltype(s)>;
590 if constexpr (!detail::is_all_v<S>) {
591 auto idx = static_cast<std::size_t>(s);
592 if (idx >= dimensions[i]) {
593 throw std::out_of_range("slice: index out of range");
594 }
595 spec_vals[i] = idx;
596 }
597 ++i;
598 }(specs),
599 ...);
600 }
601
602 auto* base = _ptr + detail::slice_helper<PaddedT>::offset(dimensions, spec_vals);
603 if constexpr (is_prefix) {
604 return ViewT{base};
605 } else {
606 return ViewT{base, detail::slice_helper<PaddedT>::strides(dimensions)};
607 }
608 }
