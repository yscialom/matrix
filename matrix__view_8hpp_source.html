583 template <typename... Specs>
584 requires(sizeof...(Specs) <= order) &&
585 ((std::same_as<Specs, all_t> || std::integral<Specs>) && ...) &&
586 (sizeof...(Specs) < order || (std::same_as<Specs, all_t> || ...))
587 [[nodiscard]] constexpr auto slice(Specs... specs) const {
588 using PaddedT = detail::pad_right_with_all_t<order, Specs...>;
589 using KeptDims = typename detail::filter_kept_dims<PaddedT, Dimensions...>::type;
590 constexpr bool is_prefix = detail::is_prefix_slice_v<PaddedT>;
591 using Storage = std::conditional_t<is_prefix, contiguous, strided>;
592 using ViewT = detail::make_matrix_view_t<T, Storage, KeptDims>;
593
594 std::array<std::size_t, order> spec_vals{};
595 {
596 std::size_t i = 0;
597 (
598 [&](auto s) {
599 using S = std::remove_cvref_t<decltype(s)>;
600 if constexpr (!detail::is_all_v<S>) {
601 auto idx = static_cast<std::size_t>(s);
602 if (idx >= dimensions[i]) {
603 throw std::out_of_range("slice: index out of range");
604 }
605 spec_vals[i] = idx;
606 }
607 ++i;
608 }(specs),
609 ...);
610 }
611
612 auto* base = _ptr + detail::slice_helper<PaddedT>::offset(dimensions, spec_vals);
613 if constexpr (is_prefix) {
614 return ViewT{base};
615 } else {
616 return ViewT{base, detail::slice_helper<PaddedT>::strides(dimensions)};
617 }
618 }
