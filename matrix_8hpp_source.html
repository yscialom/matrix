1128 [[nodiscard]] const T& at(Coords... coordinates) const {
1129 const std::array<std::ptrdiff_t, sizeof...(Coords)> coords_arr = {
1130 static_cast<std::ptrdiff_t>(coordinates)...};
1131 for (std::size_t i = 0; i < sizeof...(Coords); ++i) {
1132 // NOLINTNEXTLINE(cppcoreguidelines-pro-bounds-constant-array-index)

1137 " is out of bounds for dimension " + std::to_string(i) +
1138 // NOLINTNEXTLINE(cppcoreguidelines-pro-bounds-constant-array-index)
1139 " (size=" + std::to_string(dimensions[i]) + ")");

1141 }
