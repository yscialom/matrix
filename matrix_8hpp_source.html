1449 template <class... Coords>
1450 requires integral_coordinates<Coords...>
1451 [[nodiscard]] const T& at(Coords... coordinates) const {
1452 const std::array<std::ptrdiff_t, sizeof...(Coords)> coords_arr = {
1453 static_cast<std::ptrdiff_t>(coordinates)...};
1454 for (std::size_t i = 0; i < sizeof...(Coords); ++i) {
1455 // NOLINTNEXTLINE(cppcoreguidelines-pro-bounds-constant-array-index)
1456 if (coords_arr[i] < 0 || static_cast<std::size_t>(coords_arr[i]) >= dimensions[i]) {
1457 throw std::out_of_range(
1458 // NOLINTNEXTLINE(cppcoreguidelines-pro-bounds-constant-array-index)
1459 "matrix::at: coordinate " + std::to_string(coords_arr[i]) +
1460 " is out of bounds for dimension " + std::to_string(i) +
1461 // NOLINTNEXTLINE(cppcoreguidelines-pro-bounds-constant-array-index)
1462 " (size=" + std::to_string(dimensions[i]) + ")");
1463 }
1464 }
1465 return (*this)(coordinates...);
1466 }
